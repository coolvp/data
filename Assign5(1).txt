2. Write Python functions using sqlite3 to add, view, complete, and delete tasks in the tasks table with error 
handling and proper connection management. 
 
import sqlite3 
 
def add_task(description, due_date): 
    try: 
        conn = sqlite3.connect("todo_list.db") 
        cursor = conn.cursor() 
 
        cursor.execute(""" 
            INSERT INTO tasks (description, due_date, is_completed) 
            VALUES (?, ?, ?) 
        """, (description, due_date, 0)) 
 
        conn.commit() 
 
        task_id = cursor.lastrowid 
 
        print(f"Task added successfully. Task ID: {task_id}") 
 
        conn.close() 
 
    except sqlite3.Error as e: 
        print("Database error:", e) 
 
def view_tasks(show_completed=False): 
    try: 
        conn = sqlite3.connect("todo_list.db") 
        cursor = conn.cursor() 
 
        if show_completed: 
            cursor.execute(""" 
                SELECT id, description, due_date, is_completed 
                FROM tasks 
                ORDER BY id ASC 
            """) 
        else: 
            cursor.execute(""" 
                SELECT id, description, due_date, is_completed 
                FROM tasks 
                WHERE is_completed = 0 
                ORDER BY id ASC 
            """) 
 
        tasks = cursor.fetchall() 
 
        if not tasks: 
            print("No tasks found.") 
        else: 
            print("\n----- Tasks -----")

3. Create a Python main program loop using a menu-driven interface to add, view, complete, and delete tasks 
from a To-Do List Manager. 
 
from Assign5_Q2 import add_task, view_tasks, mark_task_completed, delete_task 
while True: 
    print("\n==========================") 
    print("     To-Do List Manager") 
    print("==========================") 
    print("1. Add New Task") 
    print("2. View Pending Tasks") 
    print("3. View All Tasks (including completed)") 
    print("4. Mark Task as Completed") 
    print("5. Delete Task") 
    print("6. Exit") 
 
    choice = input("Enter your choice: ") 
 
    if choice == "1": 
        description = input("Enter task description: ") 
        due_date = input("Enter due date (optional): ") 
 
        add_task(description, due_date) 
 
    elif choice == "2": 
        view_tasks(False) 
 
    elif choice == "3": 
        view_tasks(True) 
 
    elif choice == "4": 
        try: 
            task_id = int(input("Enter task ID: ")) 
            mark_task_completed(task_id) 
 
        except ValueError: 
            print("Please enter a valid task ID.") 
 
    elif choice == "5": 
        try: 
            task_id = int(input("Enter task ID: ")) 
            delete_task(task_id) 
 
        except ValueError: 
            print("Please enter a valid task ID.") 
 
    elif choice == "6": 
        print("Thank you for using To-Do List Manager.") 
        print("Goodbye!") 
        break 
 
    else: 
        print("Invalid choice. Please try again.")
