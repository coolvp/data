QUESTION 1: Connect to MongoDB and Select Database / Collection
--------------------------------------------------------------------------------
Task:
Write a Python script using pymongo to connect to MongoDB, handle connection errors,
and select the university_mongo_assignment database and students_data collection.

Python Script:
--------------------------------------------------------------------------------
from pymongo import MongoClient
from pymongo.errors import ConnectionFailure

MONGO_URI = "mongodb://localhost:27017/"

try:
    client = MongoClient(
        MONGO_URI,
        serverSelectionTimeoutMS=5000
    )
    client.admin.command("ping")
    print("MongoDB connected successfully!")
except ConnectionFailure as e:
    print("MongoDB connection failed:", e)
    exit()

db = client["university_mongo_assignment"]
students_collection = db["students_data"]

print("Database:", db.name)
print("Collection:", students_collection.name)

Write a Python script to perform basic, filtered, sorted, paginated, and

from pymongo import MongoClient

client = MongoClient("mongodb://localhost:27017/")
db = client["university_mongo_assignment"]
students_collection = db["students_data"]

print("\n========== 3.1 ALL STUDENTS ==========")
for student in students_collection.find():
    print(student)
print("Total number of students:", students_collection.count_documents({}))

print("\n========== 3.2 STUDENT BY EMAIL ==========")
email = "alice.s@example.com"
print(students_collection.find_one({"email": email}))

print("\n========== 3.3 COMPUTER SCIENCE STUDENTS ==========")
for student in students_collection.find({"major": "Computer Science"}):
    print(student)

print("\n========== 3.4 STUDENTS ENROLLED IN CS101 ==========")
for student in students_collection.find({"enrollments.course_code": "CS101"}):
    print(student)

print("\n========== 3.5 STUDENTS WITH PENDING MARKS ==========")
pipeline = [
    {"$unwind": "$enrollments"},
    {"$match": {"enrollments.marks": "Pending"}},
    {"$project": {
        "_id": 0,
        "name": 1,
        "email": 1,
        "course_code": "$enrollments.course_code",
        "title": "$enrollments.title"
    }}
]
for student in students_collection.aggregate(pipeline):
    print(student)

print("\n========== 3.6 TOP 3 STUDENTS ==========")
for student in students_collection.find().sort("name", 1).limit(3):
    print(student)

print("\n========== 3.7 PAGINATION ==========")
for student in students_collection.find().sort("name", 1).skip(2).limit(3):
    print(student)

print("\n========== 3.8 STUDENTS PER MAJOR ==========")
pipeline = [
    {"$group": {
        "_id": "$major",
        "total_students": {"$sum": 1}
    }},
    {"$sort": {"_id": 1}}
]
for result in students_collection.aggregate(pipeline):
    print("Major:", result["_id"], "| Count:", result["total_students"])

print("\n========== 3.9 AVERAGE MARKS FOR PH203 ==========")
pipeline = [
    {"$unwind": "$enrollments"},
    {"$match": {
        "enrollments.course_code": "PH203",
        "enrollments.marks": {"$type": "number"}
    }},
    {"$group": {
        "_id": "$enrollments.course_code",
        "average_marks": {"$avg": "$enrollments.marks"}
    }}
]
for result in students_collection.aggregate(pipeline):
    print("Course:", result["_id"])
    print("Average Marks:", result["average_marks"])

print("\n========== 3.10 STUDENTS WITH MARKS ABOVE 90 ==========")
pipeline = [
    {"$unwind": "$enrollments"},
    {"$match": {"enrollments.marks": {"$gt": 90}}},
    {"$project": {
        "_id": 0,
        "name": 1,
        "course_code": "$enrollments.course_code",
        "marks": "$enrollments.marks"
    }}
]
for student in students_collection.aggregate(pipeline):
    print(student)

client.close()

QUESTION 4: Update Operations ($set, $push, $pull, upsert)
--------------------------------------------------------------------------------
Task:
Write a Python script to perform update operations on the students_data collection, 
including modifying student details, adding/updating/removing enrollments, and 
performing an insert operation.

Python Script:
--------------------------------------------------------------------------------
from pymongo import MongoClient

client = MongoClient("mongodb://localhost:27017/")
db = client["university_mongo_assignment"]
students_collection = db["students_data"]

# 4.1 Update Major
result = students_collection.update_one(
    {"email": "alice.s@example.com"},
    {
        "$set": {
            "major": "Data Science"
        }
    }
)
print("4.1 - Alice major updated")
print("Modified documents:", result.modified_count)

# 4.2 Push New Enrollment
new_course = {
    "course_code": "CS205",
    "title": "Database Management",
    "marks": 90
}
result = students_collection.update_one(
    {"name": "Alice Smith"},
    {
        "$push": {
            "enrollments": new_course
        }
    }
)
print("\n4.2 - New enrollment added to Alice")
print("Modified documents:", result.modified_count)

# 4.3 Update Array Element
result = students_collection.update_one(
    {
        "email": "bob.b@example.com",
        "enrollments.course_code": "MA201"
    },
    {
        "$set": {
            "enrollments.$.marks": 75
        }
    }
)
print("\n4.3 - Bob's MA201 marks updated")
print("Modified documents:", result.modified_count)

# 4.4 Pull Enrollment
result = students_collection.update_one(
    {
        "email": "diana.p@example.com"
    },
    {
        "$pull": {
            "enrollments": {
                "course_code": "PH203"
            }
        }
    }
)
print("\n4.4 - PH203 enrollment removed from Diana")
print("Modified documents:", result.modified_count)

# 4.5 Upsert Operation
email = "newstudent@example.com"
result = students_collection.update_one(
    {
        "email": email
    },
    {
        "$set": {
            "name": "New Student",
            "major": "Computer Science",
            "email": email
        },
        "$setOnInsert": {
            "enrollments": []
        }
    },
    upsert=True
)
print("\n4.5 - Upsert operation")
if result.upserted_id:
    print("New student inserted.")
    print("Inserted ID:", result.upserted_id)
else:
    print("Existing student updated.")
    print("Modified count:", result.modified_count)

client.close()

QUESTION 6: JSON Export and Import Operations
--------------------------------------------------------------------------------
Task:
Write a Python script to export Physics students to a JSON Lines file and import 
2-3 new student documents from a JSON file into the students_data collection 
while handling duplicate or insertion errors.

Python Script:
--------------------------------------------------------------------------------
from pymongo import MongoClient
import json

client = MongoClient("mongodb://localhost:27017/")
db = client["university_mongo_assignment"]
students_collection = db["students_data"]

# 6.1 Export Physics students
students = students_collection.find(
    {
        "major": "Physics"
    }
)

with open(
    "physics_students.json",
    "w",
    encoding="utf-8"
) as file:
    for student in students:
        # Convert ObjectId to string
        student["_id"] = str(student["_id"])
        # Write JSON Lines format
        file.write(
            json.dumps(student) + "\n"
        )

print("6.1 - Physics students exported successfully!")
print("File: physics_students.json")

# 6.2 Create new JSON file for import
new_students = [
    {
        "name": "Jack Miller",
        "email": "jack.m@example.com",
        "major": "Physics",
        "enrollments": [
            {
                "course_code": "PH203",
                "title": "Quantum Mechanics",
                "marks": 87
            }
        ]
    },
    {
        "name": "Karen Lee",
        "email": "karen.l@example.com",
        "major": "Computer Science",
        "enrollments": [
            {
                "course_code": "CS101",
                "title": "Intro to Programming",
                "marks": 95
            },
            {
                "course_code": "CS205",
                "title": "Database Management",
                "marks": 91
            }
        ]
    },
    {
        "name": "Leo Martin",
        "email": "leo.m@example.com",
        "major": "Arts",
        "enrollments": [
            {
                "course_code": "AR101",
                "title": "Art History",
                "marks": "Pending"
            }
        ]
    }
]

with open(
    "new_students_for_import.json",
    "w",
    encoding="utf-8"
) as file:
    for student in new_students:
        file.write(
            json.dumps(student) + "\n"
        )

print("\n6.2 - new_students_for_import.json created successfully!")

# 6.3 Import JSON data with duplicate check
students_to_insert = []

with open(
    "new_students_for_import.json",
    "r",
    encoding="utf-8"
) as file:
    for line in file:
        if line.strip():
            student = json.loads(line)

            # Check if student already exists
            existing_student = students_collection.find_one(
                {
                    "email": student["email"]
                }
            )

            if existing_student:
                print(
                    "Student already exists. Skipping:",
                    student["email"]
                )
            else:
                students_to_insert.append(student)

if students_to_insert:
    result = students_collection.insert_many(
        students_to_insert,
        ordered=False
    )
    print(
        "\n6.3 - Number of students imported:",
        len(result.inserted_ids)
    )
else:
    print("\n6.3 - No new students to import.")

client.close()
print("\nMongoDB connection closed.")
