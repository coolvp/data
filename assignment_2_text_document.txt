QUESTION 1
--------------------------------------------------------------------------------
Create a SQLite database named university_assignment.db with students, courses, 
and enrollments tables.

--- Code: Assign2 Q1.py ---
import sqlite3

con = sqlite3.connect("university_assignment.db")
cur = con.cursor()
cur.execute("PRAGMA foreign_keys = ON;")

cur.execute("""
CREATE TABLE IF NOT EXISTS students (
    student_id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE,
    major TEXT
);
""")

cur.execute("""
CREATE TABLE IF NOT EXISTS courses (
    course_code TEXT PRIMARY KEY,
    title TEXT NOT NULL,
    credits INTEGER
);
""")

cur.execute("""
CREATE TABLE IF NOT EXISTS enrollments (
    enrollment_id INTEGER PRIMARY KEY AUTOINCREMENT,
    student_id INTEGER NOT NULL,
    course_code TEXT NOT NULL,
    grade TEXT,
    FOREIGN KEY (student_id) REFERENCES students(student_id),
    FOREIGN KEY (course_code) REFERENCES courses(course_code)
);
""")

print("Database and tables created successfully.")

import sqlite3

con = sqlite3.connect("university_assignment.db")
cur = con.cursor()

print("===== ALL STUDENTS =====")
cur.execute("SELECT * FROM students")
students = cur.fetchall()
for student in students:
    print(student)

print("\n===== ALL COURSES =====")
cur.execute("SELECT * FROM courses")
courses = cur.fetchall()
for course in courses:
    print(course)

print("\n===== STUDENT BY EMAIL =====")
email = "rahul@gmail.com"
cur.execute("SELECT * FROM students WHERE email = ?", (email,))
student = cur.fetchone()
if student:
    print(student)
else:
    print("Student not found.")

print("\n===== STUDENTS OF COMPUTER SCIENCE =====")
major = "Computer Science"
cur.execute("SELECT * FROM students WHERE major = ?", (major,))
records = cur.fetchall()
for record in records:
    print(record)

print("\n===== COURSES OF STUDENT (Student ID = 1) =====")
cur.execute("""
SELECT students.name,
       courses.course_code,
       courses.title,
       enrollments.grade
FROM enrollments
JOIN students ON enrollments.student_id = students.student_id
JOIN courses ON enrollments.course_code = courses.course_code
WHERE students.student_id = ?
""", (1,))
records = cur.fetchall()
for record in records:
    print(record)

print("\n===== STUDENTS ENROLLED IN CS101 =====")
cur.execute("""
SELECT students.name,
       students.email
FROM enrollments
JOIN students ON enrollments.student_id = students.student_id
WHERE enrollments.course_code = ?
""", ("CS101",))
records = cur.fetchall()
for record in records:
    print(record)

print("\n===== STUDENTS WITH PENDING GRADE =====")
cur.execute("""
SELECT students.name,
       students.email,
       enrollments.course_code
FROM enrollments
JOIN students ON enrollments.student_id = students.student_id
WHERE enrollments.grade = 'Pending'
""")
records = cur.fetchall()
for record in records:
    print(record)

print("\n===== STUDENT COUNT PER MAJOR =====")
cur.execute("""
SELECT major, COUNT(*) AS "Total Students"
FROM students
GROUP BY major
""")
records = cur.fetchall()
for record in records:
    print(record)

con.close()

Write a Python program to update student and enrollment records in the SQLite 
database and commit the changes.

--- Code: Assign2 Q4.py ---
import sqlite3

con = sqlite3.connect("university_assignment.db")
cur = con.cursor()
cur.execute("PRAGMA foreign_keys = ON;")

print("===== UPDATE STUDENT MAJOR =====")
student_name = "Rahul Sharma"
new_major = "Data Science"

cur.execute("""
UPDATE students
SET major = ?
WHERE name = ?
""", (new_major, student_name))

print("Rows affected:", cur.rowcount)
con.commit()
print("Student major updated successfully.\n")

cur.execute("SELECT * FROM students WHERE name = ?", (student_name,))
print(cur.fetchone())

print("\n===== UPDATE ENROLLMENT GRADE =====")
student_id = 2
course_code = "CS101"
new_grade = "B+"

cur.execute("""
UPDATE enrollments
SET grade = ?
WHERE student_id = ? AND course_code = ?
""", (new_grade, student_id, course_code))

print("Rows affected:", cur.rowcount)
con.commit()
print("Enrollment grade updated successfully.\n")

cur.execute("""
SELECT *
FROM enrollments
WHERE student_id = ? AND course_code = ?
""", (student_id, course_code))
print(cur.fetchone())

con.close()

Write a Python program to perform delete operations in the SQLite database 
while handling foreign key constraints.

--- Code: Assign2 Q5.py ---
import sqlite3

con = sqlite3.connect("university_assignment.db")
cur = con.cursor()
cur.execute("PRAGMA foreign_keys = ON;")

print("===== DELETE AN ENROLLMENT =====")
student_id = 5
course_code = "DB301"

cur.execute("""
DELETE FROM enrollments
WHERE student_id = ? AND course_code = ?
""", (student_id, course_code))

print("Rows affected:", cur.rowcount)
con.commit()
print("Enrollment deleted successfully.\n")

print("===== ATTEMPT TO DELETE ENROLLED STUDENT =====")
try:
    cur.execute("""
    DELETE FROM students
    WHERE student_id = ?
    """, (1,))
    con.commit()
except sqlite3.IntegrityError as e:
    print("Error:", e)

print()
print("===== SAFE STUDENT DELETION =====")
student_id = 4

cur.execute("""
DELETE FROM enrollments
WHERE student_id = ?
""", (student_id,))

print("Enrollment rows deleted:", cur.rowcount)

cur.execute("""
DELETE FROM students
WHERE student_id = ?
""", (student_id,))

print("Student rows deleted:", cur.rowcount)
con.commit()
print("Student and related enrollments deleted successfully.")

con.close()

QUESTION 6

Write a Python program to create full, schema-only, and data-only dumps of 
the SQLite database into SQL files.

--- Code: Assign2 Q6.py ---
import sqlite3

con = sqlite3.connect("university_assignment.db")
cur = con.cursor()

with open("assignment_full_dump.sql", "w") as f:
    for line in con.iterdump():
        f.write("%s\n" % line)

print("Full database dump created: assignment_full_dump.sql")

with open("assignment_schema_only.sql", "w") as f:
    cur.execute("""
    SELECT sql
    FROM sqlite_master
    WHERE type='table'
    AND name IN ('students', 'enrollments')
    """)
    tables = cur.fetchall()
    for table in tables:
        f.write(table[0] + ";\n\n")

print("Schema-only dump created: assignment_schema_only.sql")

with open("courses_data_dump.sql", "w") as f:
    cur.execute("SELECT * FROM courses")
    records = cur.fetchall()
    for record in records:
        sql = "INSERT INTO courses (course_code, title, credits) VALUES ('{}', '{}', {});\n".format(
            record[0], record[1], record[2]
        )
        f.write(sql)

print("Data-only dump created: courses_data_dump.sql")
con.close()

QUESTION 7
--------------------------------------------------------------------------------
Write a Python program to export enrollment data to a CSV file and import 
new student records from a CSV file into the SQLite database.

--- Code: Assign2 Q7.py ---
import sqlite3
import csv

con = sqlite3.connect("university_assignment.db")
cur = con.cursor()

cur.execute("""
SELECT
    enrollments.enrollment_id,
    students.name,
    courses.title,
    enrollments.grade
FROM enrollments
JOIN students ON enrollments.student_id = students.student_id
JOIN courses ON enrollments.course_code = courses.course_code
""")

records = cur.fetchall()

with open("enrollments_detailed_export.csv", "w", newline="") as file:
    writer = csv.writer(file)
    writer.writerow(["Enrollment ID", "Student Name", "Course Title", "Grade"])
    writer.writerows(records)

print("Enrollment data exported successfully to enrollments_detailed_export.csv")

with open("new_students_to_import.csv", "r") as file:
    reader = csv.reader(file)
    # Skip Header Row
    next(reader)
    for row in reader:
        cur.execute("""
        INSERT INTO students(name, email, major)
        VALUES (?, ?, ?)
        """, row)

con.commit()
print("New students imported successfully.")

print("\nUpdated Students Table:\n")
cur.execute("SELECT * FROM students")
records = cur.fetchall()
for record in records:
    print(record)

con.close()
